## Model name
Prophet

## Example Python code
from prophet import Prophet
import pandas as pd

# Sample data (replace with your actual data)
data = pd.DataFrame({ 
    'ds': pd.to_datetime(['2024-06-07', '2024-06-08', '2024-06-09', '2024-06-10', '2024-06-11']),
    'y': [10, 12, 15, 13, 17]
})

model = Prophet()
model.fit(data)

# Create future dataframe for forecasting
future = model.make_future_dataframe(periods=7) # forecasting 7 days
forecast = model.predict(future)

print(forecast[['ds', 'yhat', 'yhat_lower', 'yhat_upper']].tail(7))
